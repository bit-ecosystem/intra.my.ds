namespace YourVendor\Workflow\Services;

use YourVendor\Workflow\Models\Workflow;

class WorkflowResolver
{
    public static function resolve($model): ?Workflow
    {
        return Workflow::query()
            ->where('model_type', get_class($model))
            ->where('is_active', true)
            ->orderByDesc('priority')
            ->first();
    }
}

namespace YourVendor\Workflow\Traits;

use Exception;
use RingleSoft\ProcessApproval\Enums\ApprovalTypeEnum;
use YourVendor\Workflow\Services\WorkflowResolver;

trait HasWorkflow
{
    public function startWorkflow()
    {
        $workflow = WorkflowResolver::resolve($this);

        if (!$workflow) {
            throw new Exception('No workflow configured.');
        }

        $steps = [];

        foreach ($workflow->steps as $step) {
            $steps[$step->step_order] =
                $step->type === 'check'
                    ? ApprovalTypeEnum::CHECK
                    : ApprovalTypeEnum::APPROVE;
        }

        static::makeApprovable($steps);

        $this->submit();
    }
}

namespace YourVendor\Workflow\Filament\Resources;

use Filament\Forms;
use Filament\Resources\Resource;
use Filament\Tables;
use YourVendor\Workflow\Models\Workflow;

class WorkflowResource extends Resource
{
    protected static ?string $model = Workflow::class;

    public static function form(Forms\Form $form): Forms\Form
    {
        return $form->schema([
            Forms\Components\TextInput::make('name')->required(),

            Forms\Components\TextInput::make('model_type')
                ->required(),

            Forms\Components\TextInput::make('document_type'),

            Forms\Components\TextInput::make('priority')
                ->numeric()
                ->default(1),

            Forms\Components\Toggle::make('is_active'),

            Forms\Components\Repeater::make('steps')
                ->relationship()
                ->schema([
                    Forms\Components\TextInput::make('step_order')
                        ->numeric()
                        ->required(),

                    Forms\Components\TextInput::make('role')
                        ->required(),

                    Forms\Components\Select::make('type')
                        ->options([
                            'check' => 'Check',
                            'approve' => 'Approve',
                        ])
                        ->required(),

                    Forms\Components\Toggle::make('is_required'),
                ])
                ->orderable('step_order')
        ]);
    }

    public static function table(Tables\Table $table): Tables\Table
    {
        return $table->columns([
            Tables\Columns\TextColumn::make('name'),
            Tables\Columns\TextColumn::make('model_type'),
            Tables\Columns\TextColumn::make('priority'),
            Tables\Columns\IconColumn::make('is_active')->boolean(),
        ]);
    }

    public static function getPages(): array
    {
        return [
            'index' => Pages\ListWorkflows::route('/'),
            'create' => Pages\CreateWorkflow::route('/create'),
            'edit' => Pages\EditWorkflow::route('/{record}/edit'),
        ];
    }
}
